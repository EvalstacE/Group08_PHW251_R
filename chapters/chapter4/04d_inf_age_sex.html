13  Infections by Age and Sex

When we looked at infection and severe infection rates across the state of California as a whole, females had a lower infection rate but a higher severe infection rate compared to males.

When we compared male and female infection rates by health officer region, this trend still appears.

Below is a table of infections and severe infections broken down by sex and health officer region:

by_sex <- combined_df %>%
  group_by(health_officer_region, county, sex) %>%
    summarise(
      total_sex_pop       = first(total_sex_pop),
      cumulative_infected = max(cumulative_infected, na.rm = TRUE),
      cumulative_severe   = max(cumulative_severe, na.rm = TRUE),
      inf_rate_100k       = round((10^5*cumulative_infected/total_sex_pop)),
      sev_rate_100k       = round((10^5*cumulative_severe/total_sex_pop)),
      .groups = "drop"
    ) %>% 
  group_by(health_officer_region, sex) %>%
    summarise(
        total_pop           = sum(total_sex_pop),
        cumulative_infected = sum(cumulative_infected, na.rm = TRUE),
        cumulative_severe   = sum(cumulative_severe, na.rm = TRUE),
        inf_rate_100k       = round((10^5*cumulative_infected/total_pop)),
        sev_rate_100k       = round((10^5*cumulative_severe/total_pop)),
        .groups = "drop"
    ) %>%
  arrange(health_officer_region, sex)
  

sex_tbl_df <- by_sex %>%
  rename(
    `Health Officer Region` = health_officer_region,
    Sex                     = sex,
    Population              = total_pop,
    Infections              = cumulative_infected,
    `Severe Infections`     = cumulative_severe,
    `Infection Rate (/100k)`       = inf_rate_100k,
    `Severe Rate (/100k)`    = sev_rate_100k
  ) 

sex_df_disp_grp <- sex_tbl_df %>%
  group_by(`Health Officer Region`) %>%
  mutate(
    `Health Officer Region` = if_else(row_number() == 1, `Health Officer Region`, "")
  ) %>%
  ungroup()



n_rows <- nrow(sex_df_disp_grp)
