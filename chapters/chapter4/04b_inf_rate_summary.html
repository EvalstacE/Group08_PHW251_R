##--create the kable outputs

make_state_kable <- function(df, group_name) {
  
# conditionally set group_var table label
demo_label <- dplyr::case_when(
  group_name == "sex"        ~ "Sex",
  group_name == "race_short" ~ "Race",
  group_name == "age_cat"    ~ "Age Group",
  TRUE                       ~ "Demographic"
)

# highlight highest rate per demographic group  
highlight_rows <- which(df$high_grp == "yes")
  
  df %>%
    select(
      group_var_cat,
      inf_rate_100k,
      sev_rate_100k,
      high_grp
    ) %>%
    mutate(
      inf_rate_100k = scales::comma(inf_rate_100k, accuracy = 1),
      sev_rate_100k = scales::comma(sev_rate_100k, accuracy = 1)
    ) %>%
    rename(
      !!demo_label            := group_var_cat,
      `Infection Rate`        = inf_rate_100k,
      `Severe Infection Rate` = sev_rate_100k
    ) %>%
    select(-high_grp) %>%
    
    kbl(escape = FALSE, align = "c") %>%
    
    row_spec(
      0,
      bold       = TRUE,
      background = "#0f172a",
      extra_css  = "font-size: 14px!important; color:#ffffff;"
    ) %>%
    
    row_spec(
      highlight_rows,
      background = "#fde586",
      bold       = TRUE
    ) %>%
    
    kable_styling(
      bootstrap_options = c("bordered", "condensed"),
      font_size = 12,
      full_width = FALSE
    ) %>%
    column_spec(1:3, extra_css = "padding: 2px 6px !important;")
}

cnty_ranked_df <- cnty_adj_rates %>%
  ##--compute quantile cutpoints for each metric
  mutate(
    inf_p90 = quantile(adj_rate_100k,    0.90, na.rm = TRUE),
    inf_p75 = quantile(adj_rate_100k,    0.75, na.rm = TRUE),
    inf_p50 = quantile(adj_rate_100k,    0.50, na.rm = TRUE),
    
    sev_p90 = quantile(adj_sev_100k,     0.90, na.rm = TRUE),
    sev_p75 = quantile(adj_sev_100k,     0.75, na.rm = TRUE),
    sev_p50 = quantile(adj_sev_100k,     0.50, na.rm = TRUE)
  ) %>%
  ##--assign ranks based on those cutpoints
  mutate(
    inf_q_rank = case_when(
      adj_rate_100k    >= inf_p90 ~ 1L,              # ≥ 90th
      adj_rate_100k    >= inf_p75 ~ 2L,              # 75–<90
      adj_rate_100k    >= inf_p50 ~ 3L,              # 50–<75
      TRUE                         ~ 4L              # < 50
    ),
    sev_q_rank = case_when(
      adj_sev_100k     >= sev_p90 ~ 1L,
      adj_sev_100k     >= sev_p75 ~ 2L,
      adj_sev_100k     >= sev_p50 ~ 3L,
      TRUE                         ~ 4L
    ),
    
    rank_sum = inf_q_rank + sev_q_rank,
    
    priority_tier = case_when(
      rank_sum == 2              ~ "Top Priority",      
      between(rank_sum, 3, 4)    ~ "Second Priority",  
      TRUE                       ~ "Third Priority"
    ),
    
    priority_tier = factor(
      priority_tier,
      levels = c("Top Priority", "Second Priority", "Third Priority")
    )
  ) %>%
##--drop cutpoints
  select(-starts_with("inf_p"), -starts_with("sev_p")) %>%
  arrange(priority_tier, desc(adj_sev_100k), desc(adj_rate_100k))

County Prioritization Method

To identify counties with the greatest burden of infection and severe infection, we constructed a ranking system based on age-adjusted infection rates and age-adjusted severe infection rates. We first combined two datasets. The overall county dataset included population size, total infections, total severe infections, and crude rates per 100,000. The age-adjusted dataset provided refined infection and severe infection rates that account for differences in county age distributions.

Using the age-adjusted metrics, we calculated percentile cutpoints for each rate. Counties were compared against the 90th percentile, the 75th percentile, and the 50th percentile of the statewide distribution. Counties at or above the 90th percentile were assigned a rank of 1, those between the 75th and 90th percentiles received a rank of 2, those between the 50th and 75th percentiles received a rank of 3, and the remainder received a rank of 4. This ranking process was applied to both the adjusted infection rate and the adjusted severe infection rate.

Each county received two ranks. These were summed to create a combined burden score. Counties with a score of 2 were classified as Top Priority, counties with a score of 3 or 4 were classified as Second Priority, and all remaining counties were classified as Third Priority. This approach allows us to highlight counties experiencing consistently high infection and severe infection burdens
